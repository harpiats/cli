import { AppError } from "@harpia/common";
import { Utils } from "@harpia/common";

import { repository } from "../repositories";
import type { SchemaCreateBodyType } from "../validations/create";

export async function create(data: SchemaCreateBodyType) {
  const query = await repository.create(data);

  if (!query) throw AppError.E_GENERIC_ERROR("Error trying to save data.");

  const result = Utils.object.omit(query, ["createdAt", "updatedAt"]);

  return result;
}
