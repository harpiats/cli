import { AppError } from "@harpiats/common";
import { service } from ".";
import { repository } from "../repositories";

import type { SchemaUpdateBodyType } from "../validations/update";
import type { {{ singularize(pascalCase(name)) }}Type } from "app/database";

export async function update(id: {{ singularize(pascalCase(name)) }}Type["id"], data: SchemaUpdateBodyType) {
  await service.show(id);

  const result = await repository.update(id, data);
  if (!result) throw AppError.E_GENERIC_ERROR("Error trying to save data.");

  return result;
}
